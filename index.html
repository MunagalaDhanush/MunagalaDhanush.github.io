usa_medals <- athlete_events %>%
  filter(Team == "United States",
         !is.na(Medal)) %>%
  group_by(Sport) %>%
  summarise(total = n()) %>%
  arrange(desc(total)) %>%
  top_n(12)
nations <- athlete_events %>%
  filter(!is.na(Medal)) %>%
  group_by(Team) %>%
  summarise(medals = n()) %>%
  top_n(10, medals) %>%
  ggplot(aes(x=reorder(Team,medals),
            y=medals,fill=medals)) +
  geom_bar(stat="identity") +
  coord_flip()
# 2016 Women's medals
women_2016 <- athlete_events %>%
  filter(Year==2016, Sex=="F",
         Team=="United States",
         !is.na(Medal)) %>%
  group_by(Event, Medal) %>%
  summarise(n=n())
# Decade trend analysis
decade_medals <- athlete_events %>%
  filter(!is.na(Medal)) %>%
  mutate(decade = (Year %/% 10) * 10) %>%
  group_by(decade, Team) %>%
  summarise(total=n()) %>%
  filter(Team %in% top5_nations)
